//Operadores relacionais
/* < | menor
 * > | maior
 * <= | menor ou igual
 * >= | maior ou igual
 * == | igual
 * != | diferente
 */


int x = 2;
int y = 2;
Console.WriteLine(x < y);
Console.WriteLine(x > y);
Console.WriteLine(x <= y);
Console.WriteLine(x >= y);
Console.WriteLine(x == y);
Console.WriteLine(x != y);

string a = "SENAI";
string b = "Americana";
Console.WriteLine(a == b);
Console.WriteLine(a != b);

Console.WriteLine(a == b ? "Verdade" : "OK");


//Operadores Lógicos
/*
 * ! | Negação lógica
 * & | E lógico
 * | | OU lógico
 * ^ | OU exclusiva lógica
 * && | E lógico condicional
 * || | OU lógico condicional
*/

bool x = true;
bool y = false;

Console.WriteLine(x ^ y);
